{{- /* Resolve the theme title for a project page by walking theme → project trees. Returns the first matching theme title, or empty string. */ -}} {{- $project := .page -}} {{- $projectSlug := $project.File.ContentBaseName -}} {{- $allProjects := default (where site.RegularPages "Section" "projects") .allProjects -}} {{- $themes := default (sort (where site.RegularPages "Section" "themes") "Params.weight") .themes -}} {{- $themeTitle := "" -}} {{- range $themes -}} {{- if eq $themeTitle "" -}} {{- $rootSlugs := default (slice) .Params.projects -}} {{- $match := false -}} {{- if in $rootSlugs $projectSlug -}} {{- $match = true -}} {{- else -}} {{- range $rootSlugs -}} {{- $descRaw := partial "collect-descendant-project-slugs.html" (dict "slug" . "allProjects" $allProjects) -}} {{- if and (ne $descRaw "") (in (split $descRaw "\n") $projectSlug) -}} {{- $match = true -}} {{- end -}} {{- end -}} {{- end -}} {{- if $match -}} {{- $themeTitle = .Title -}} {{- end -}} {{- end -}} {{- end -}} {{- return $themeTitle -}}
