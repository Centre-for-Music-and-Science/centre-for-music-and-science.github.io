"string" -}} {{- errorf "Topic error: topic %q has a cosupervisor slug that is not a string." $slug -}} {{- else if eq (strings.TrimSpace .) "" -}} {{- errorf "Topic error: topic %q has an empty cosupervisor slug." $slug -}} {{- else if not (in $peopleSlugs .) -}} {{- errorf "Topic error: topic %q links to unknown person %q." $slug . -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- if and (isset .Params "projects") (not (reflect.IsSlice .Params.projects)) -}} {{- errorf "Topic error: topic %q has non-list projects." $slug -}} {{- end -}} {{- range (default (slice) .Params.projects) -}} {{- if not (in $projectSlugs .) -}} {{- errorf "Topic error: topic %q links to unknown project %q." $slug . -}} {{- end -}} {{- end -}} {{- if and (isset .Params "publications") (not (reflect.IsSlice .Params.publications)) -}} {{- errorf "Topic error: topic %q has non-list publications." $slug -}} {{- end -}} {{- range (default (slice) .Params.publications) -}} {{- if not (in $publicationSlugs .) -}} {{- errorf "Topic error: topic %q links to unknown publication %q." $slug . -}} {{- end -}} {{- end -}} {{- end -}}
